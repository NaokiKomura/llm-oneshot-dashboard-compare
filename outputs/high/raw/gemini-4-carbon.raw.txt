「ゲーム」領域を題材に、**マクロ（市場全体）からミクロ・ナノ（個別タイトル：マリオカートワールド）までシームレスに粒度を遷移（ドリルダウン／ロールアップ）**しながら、発話量推移・トピック内訳・評価軸別センチメント・共起キーワード・AIインサイト・実際のXポスト（ダミー）を多面的に可視化するソーシャルリスニング・ダッシュボードのHTMLモックアップを作成しました。

以下のコードを `.html` ファイル（例: `x_topic_dashboard.html`）として保存し、ブラウザで開くだけで全てのインタラクションが動作します。

### 主な実装機能
1. **4階層の粒度遷移（マクロ〜ナノ）**
   - 左サイドバーの階層ツリー、上部のパンくずリスト、または**ドーナツグラフのクリック**によって直感的にドリルダウン可能。
   - 例：`Lv.1 ゲーム全般` ➔ `Lv.2 コンシューマーゲーム` ➔ `Lv.3 任天堂スイッチ` ➔ `Lv.4 マリオカート ワールド`（他分岐としてスマホ、PC、PS5、ゼルダ等も収録）
   - 右上の「デモ：1段階深掘り」ボタンで、推奨シナリオに沿った段階的な粒度遷移をワンクリックで体験できます。
2. **多面的なデータ可視化**
   - **日次発話量×感情推移グラフ**：単なる件数だけでなく、スパイク（急増）が「歓喜（ポジティブ）」か「炎上・不満（ネガティブ）」かを積み上げ棒＋折れ線で可視化し、スパイク要因も自動表示。
   - **構成トピック内訳**：現在の粒度におけるサブカテゴリ比率を表示（クリックで下位粒度へ遷移、最下層ではポストをフィルタリング）。
   - **評価軸（アスペクト）別ポジネガ分析**：価格、ゲーム性、スペックなど論点ごとの賛否バランスを可視化。
   - **AI構造化インサイト**：各粒度における「発話背景」「生活者の本音・インサイト」「示唆・アクション」を提示。
   - **共起キーワード＆Xポストストリーム**：感情スコア付きトレンドワードと、AI注釈付きのリアルなダミーXポスト（感情・バズ別フィルタ対応）。

    function navigateTo(topicId) {
      if (!TOPIC_DATA[topicId]) return;
      currentTopicId = topicId;

    // ==================== SIDEBAR & TREE RENDERING ====================
    function renderSidebarTree() {
      const current = TOPIC_DATA[currentTopicId];
      const level = current.level;

      // Update Level Badge & Progress Meter
      const levelNames = { 1: 'Lv.1 マクロ', 2: 'Lv.2 ミドル', 3: 'Lv.3 ミクロ', 4: 'Lv.4 ナノ' };
      document.getElementById('sidebarLevelBadge').textContent = levelNames[level];

      for (let i = 1; i <= 4; i++) {
        const bar = document.getElementById(`meterLv${i}`);
        if (i <= level) {
          bar.className = `h-full flex-1 transition-all duration-300 ${i === 1 ? 'rounded-l-full' : ''} ${i === 4 ? 'rounded-r-full' : ''} ${
            level === 4 ? 'bg-amber-400' : 'bg-indigo-500'
          }`;
        } else {
          bar.className = `h-full flex-1 bg-slate-700 transition-all duration-300 ${i === 4 ? 'rounded-r-full' : ''}`;
        }
      }

      // Recursive Tree Builder
      const container = document.getElementById('topicTreeContainer');
      container.innerHTML = buildTreeHTML('game', 0);
    }

    function buildTreeHTML(nodeId, depth) {
      const node = TOPIC_DATA[nodeId];
      const isSelected = node.id === currentTopicId;
      const isMainDemoPath = DEMO_PATH.includes(node.id);

      const indentClass = depth === 0 ? '' : depth === 1 ? 'ml-3 pl-2.5 border-l border-slate-700/80' : depth === 2 ? 'ml-6 pl-2.5 border-l border-slate-700/80' : 'ml-9 pl-2.5 border-l border-slate-700/80';

      const icon = node.level === 1 ? 'fa-globe' :
                   node.level === 2 ? 'fa-layer-group' :
                   node.level === 3 ? 'fa-gamepad' : 'fa-flag-checkered';

      const activeClasses = isSelected
        ? 'bg-indigo-600 text-white font-semibold shadow-sm'
        : 'text-slate-300 hover:bg-slate-800 hover:text-white';

      let html = `
        <div class="${indentClass} my-0.5">
          <button onclick="navigateTo('${node.id}')" class="w-full text-left px-2.5 py-1.5 rounded-lg text-xs flex items-center justify-between transition group ${activeClasses}">
            <div class="flex items-center gap-2 truncate">
              <i class="fa-solid ${icon} text-[11px] ${isSelected ? 'text-indigo-200' : 'text-slate-500 group-hover:text-slate-300'}"></i>
              <span class="truncate">${node.shortName}</span>
            </div>
            <div class="flex items-center gap-1 flex-shrink-0">
              ${isMainDemoPath && !isSelected ? '<span class="w-1.5 h-1.5 rounded-full bg-sky-400" title="推奨デモ経路"></span>' : ''}
              <span class="text-[10px] ${isSelected ? 'text-indigo-200' : 'text-slate-500'}">Lv.${node.level}</span>
            </div>
          </button>
        </div>
      `;

      if (node.children && node.children.length > 0) {
        node.children.forEach(childId => {
          html += buildTreeHTML(childId, depth + 1);
        });
      }
      return html;
    }

    // ==================== HEADER, BREADCRUMBS & PILLS ====================
    function renderHeaderAndBreadcrumbs() {
      const current = TOPIC_DATA[currentTopicId];

      // Roll-up button state
      document.getElementById('btnRollUp').disabled = !current.parent;

      // Build ancestor path
      const path = [];
      let ptr = current;
      while (ptr) {
        path.unshift(ptr);
        ptr = ptr.parent ? TOPIC_DATA[ptr.parent] : null;
      }

      // Render Breadcrumbs
      const nav = document.getElementById('breadcrumbNav');
      nav.innerHTML = path.map((item, idx) => {
        const isLast = idx === path.length - 1;
        if (isLast) {
          return `<span class="px-2.5 py-1 rounded-md bg-indigo-50 text-indigo-700 font-bold border border-indigo-200/70">${item.name}</span>`;
        }
        return `
          <button onclick="navigateTo('${item.id}')" class="px-2 py-1 rounded-md text-slate-600 hover:text-indigo-600 hover:bg-slate-100 transition">
            ${item.shortName}
          </button>
          <i class="fa-solid fa-chevron-right text-[10px] text-slate-400"></i>
        `;
      }).join('');

      // Update Demo Stepper Button Text
      const demoIdx = DEMO_PATH.indexOf(currentTopicId);
      const btnText = document.getElementById('btnDemoStepText');
      if (demoIdx === -1 || demoIdx < DEMO_PATH.length - 1) {
        const nextId = demoIdx === -1 ? DEMO_PATH[1] : DEMO_PATH[demoIdx + 1];
        const nextNode = TOPIC_DATA[nextId];
        btnText.textContent = `デモ：『${nextNode.shortName}』へ深掘り (Lv.${nextNode.level})`;
      } else {
        btnText.textContent = `デモ完了：Lv.1『ゲーム全般』へ戻る`;
      }

      // Banner Info
      const badgeColors = {
        1: 'bg-indigo-100 text-indigo-700',
        2: 'bg-sky-100 text-sky-700',
        3: 'bg-emerald-100 text-emerald-700',
        4: 'bg-amber-100 text-amber-800'
      };
      const headerBadge = document.getElementById('headerGranularityBadge');
      headerBadge.className = `px-2.5 py-0.5 rounded-md text-xs font-bold ${badgeColors[current.level]}`;
      headerBadge.textContent = current.levelLabel;

      document.getElementById('headerCategoryPath').textContent = path.map(p => p.shortName).join(' ➔ ');
      document.getElementById('currentTopicTitle').textContent = current.name;
      document.getElementById('currentTopicDesc').textContent = current.desc;

      // Quick Child Drill-down Pills
      const childArea = document.getElementById('childDrilldownArea');
      if (current.children && current.children.length > 0) {
        const pills = current.children.map(cid => {
          const child = TOPIC_DATA[cid];
          const highlight = DEMO_PATH.includes(cid) ? 'border-indigo-300 bg-indigo-50/70 hover:bg-indigo-100 text-indigo-700' : 'border-slate-200 bg-slate-50 hover:bg-slate-100 text-slate-700';
          return `
            <button onclick="navigateTo('${child.id}')" class="px-3 py-1.5 rounded-xl border text-xs font-semibold flex items-center gap-1.5 transition shadow-sm ${highlight}">
              <i class="fa-solid fa-magnifying-glass-plus text-[11px]"></i>
              ${child.shortName}
              <i class="fa-solid fa-arrow-right text-[10px] opacity-70"></i>
            </button>
          `;
        }).join('');
        childArea.innerHTML = `
          <span class="text-[11px] text-slate-400 font-medium">さらにミクロな粒度へドリルダウン：</span>
          <div class="flex flex-wrap gap-2">${pills}</div>
        `;
      } else {
        childArea.innerHTML = `
          <div class="px-3 py-2 rounded-xl bg-amber-50 border border-amber-200 text-amber-800 text-xs flex items-center gap-2">
            <i class="fa-solid fa-bullseye text-amber-600"></i>

    function handleSubtopicClick(index) {
      const item = TOPIC_DATA[currentTopicId].subtopics[index];
      if (!item) return;

      if (item.targetId) {
        navigateTo(item.targetId);
      } else if (item.filterTopic) {
        currentSubtopicFilter = item.filterTopic;
        currentKeywordFilter = null;
        renderKeywords();
        renderPosts();
        document.getElementById('xPostsContainer').scrollIntoView({ behavior: 'smooth', block: 'nearest' });
      }
    }

    // ==================== KEYWORDS CLOUD ====================
    function renderKeywords() {
      const d = TOPIC_DATA[currentTopicId];
      const container = document.getElementById('keywordCloud');
      const resetBtn = document.getElementById('btnResetKeyword');

      resetBtn.classList.toggle('hidden', !currentKeywordFilter);

      container.innerHTML = d.keywords.map(k => {
        const isSelected = currentKeywordFilter === k.word;
        const sizeClass = k.weight === 5 ? 'text-sm font-bold px-3 py-1.5' :
                          k.weight === 4 ? 'text-xs font-semibold px-2.5 py-1' :
                          'text-xs font-medium px-2 py-1';

        let colorClass = '';
        if (isSelected) {
          colorClass = 'bg-indigo-600 text-white border-indigo-600 ring-2 ring-indigo-200';
        } else if (k.sentiment === 'pos') {
          colorClass = 'bg-emerald-50 text-emerald-700 border-emerald-200 hover:bg-emerald-100';
        } else if (k.sentiment === 'neg') {
          colorClass = 'bg-rose-50 text-rose-700 border-rose-200 hover:bg-rose-100';
        } else {
          colorClass = 'bg-sky-50 text-sky-700 border-sky-200 hover:bg-sky-100';
        }

        return `
          <button onclick="setKeywordFilter('${k.word}')" class="rounded-lg border transition ${sizeClass} ${colorClass}">
            ${k.word.startsWith('#') ? k.word : '#' + k.word}
          </button>
        `;
      }).join('');
    }

    function setKeywordFilter(word) {
      currentKeywordFilter = (currentKeywordFilter === word) ? null : word;
      currentSubtopicFilter = null;
      renderKeywords();
      renderPosts();
    }

    function updateFilterTabsUI() {
      document.querySelectorAll('.post-filter-btn').forEach(btn => {
        const active = btn.dataset.filter === currentSentimentFilter;
        btn.className = active
          ? 'post-filter-btn px-3 py-1 rounded-lg bg-white text-slate-900 shadow-sm font-semibold transition'
          : 'post-filter-btn px-3 py-1 rounded-lg text-slate-600 hover:text-slate-900 transition';
      });
    }

    function renderPosts() {
      const d = TOPIC_DATA[currentTopicId];
      let posts = [...d.posts];

      // Apply Sentiment / Buzz filter
      if (currentSentimentFilter === 'positive') {
        posts = posts.filter(p => p.sentiment === 'positive');
      } else if (currentSentimentFilter === 'negative') {
        posts = posts.filter(p => p.sentiment === 'negative');
      } else if (currentSentimentFilter === 'buzz') {
        posts = posts.filter(p => p.buzz);
      }

      // Apply Keyword filter
      if (currentKeywordFilter) {
        const cleanKw = currentKeywordFilter.replace('#', '');
        posts = posts.filter(p =>
          p.text.includes(cleanKw) || p.tags.some(t => t.includes(cleanKw) || cleanKw.includes(t))
        );
      }

      // Apply Subtopic filter (on leaf nodes)
      if (currentSubtopicFilter) {
        posts = posts.filter(p =>
          p.text.includes(currentSubtopicFilter) || p.tags.some(t => t.includes(currentSubtopicFilter))
        );
      }

      // Active Filter Bar visibility
      const filterBar = document.getElementById('activePostFilterBar');
      const filterText = document.getElementById('activePostFilterText');
      if (currentKeywordFilter || currentSubtopicFilter) {
        filterBar.classList.remove('hidden');
        const label = currentKeywordFilter ? `キーワード「${currentKeywordFilter}」` : `論点トピック「${currentSubtopicFilter}」`;
        filterText.innerHTML = `<i class="fa-solid fa-filter mr-1.5"></i>${label} に関連するポストを表示中 (${posts.length}件)`;
      } else {
        filterBar.classList.add('hidden');
      }

      document.getElementById('postStreamSubtitle').textContent = `『${d.name}』粒度における代表的なX発話 (${posts.length}件表示)`;

      const container = document.getElementById('xPostsContainer');
      if (posts.length === 0) {
        container.innerHTML = `
          <div class="py-12 text-center text-slate-400 space-y-2">
            <i class="fa-regular fa-comment-dots text-2xl"></i>
            <p class="text-xs">該当条件に一致する代表的なダミーポストがありません。</p>
            <button onclick="clearAllPostFilters()" class="text-xs text-indigo-600 font-semibold hover:underline">フィルタをリセット</button>
          </div>
        `;
        return;
      }

      container.innerHTML = posts.map(p => {
        const isPos = p.sentiment === 'positive';
        const sentimentBadge = isPos
          ? '<span class="px-2 py-0.5 rounded-full text-[10px] font-semibold bg-emerald-50 text-emerald-700 border border-emerald-200">Positive</span>'
          : '<span class="px-2 py-0.5 rounded-full text-[10px] font-semibold bg-rose-50 text-rose-700 border border-rose-200">Concern / Neg</span>';

        const buzzBadge = p.buzz
          ? '<span class="px-2 py-0.5 rounded-full text-[10px] font-semibold bg-amber-50 text-amber-700 border border-amber-200"><i class="fa-solid fa-fire mr-1"></i>拡散バズ</span>'
          : '';

        const avatarBg = isPos ? 'from-indigo-500 to-sky-400' : 'from-rose-500 to-amber-500';
        const initial = p.user.charAt(0);

        return `
          <article class="p-4 rounded-xl border border-slate-200/90 hover:border-slate-300 bg-slate-50/40 transition space-y-2.5">
            <!-- User Header -->
            <div class="flex items-start justify-between gap-2">
              <div class="flex items-center gap-2.5">
                <div class="w-9 h-9 rounded-full bg-gradient-to-br ${avatarBg} text-white font-bold flex items-center justify-center text-sm shadow-sm flex-shrink-0">
                  ${initial}
                </div>
                <div>
                  <div class="flex items-center gap-1.5">
                    <span class="font-bold text-slate-900 text-xs">${p.user}</span>
                    <span class="text-[11px] text-slate-400">${p.handle} · ${p.time}</span>
                  </div>
                  <div class="flex flex-wrap gap-1 mt-0.5">
                    ${p.tags.slice(0, 3).map(t => `<span class="text-[10px] text-indigo-600 bg-indigo-50 px-1.5 py-0.2 rounded">#${t.replace('#','')}</span>`).join('')}
                  </div>
                </div>
              </div>
              <div class="flex items-center gap-1.5 flex-shrink-0">
                ${buzzBadge}
                ${sentimentBadge}
              </div>
            </div>

            <!-- Post Body -->
            <p class="text-xs text-slate-800 leading-relaxed pl-11">
              ${p.text}
            </p>

            <!-- Engagement Metrics -->
            <div class="pl-11 flex items-center gap-5 text-[11px] text-slate-400">
              <span><i class="fa-solid fa-retweet mr-1 text-emerald-500"></i>${p.rts.toLocaleString()}</span>
              <span><i class="fa-regular fa-heart mr-1 text-rose-500"></i>${p.likes.toLocaleString()}</span>
            </div>

            <!-- AI Micro-Insight Annotation -->
            <div class="ml-11 p-2.5 rounded-lg bg-indigo-50/70 border border-indigo-100 text-[11px] text-indigo-950 flex items-start gap-2">
              <i class="fa-solid fa-lightbulb text-indigo-600 mt-0.5"></i>
              <div><strong class="font-semibold text-indigo-700">抽出インサイト：</strong>${p.aiNote}</div>
            </div>
          </article>
        `;
      }).join('');
    }
