マクロからミクロへ段階的にドリルダウンしながら、X（旧Twitter）の発話傾向・トピック・インサイトを多面的に分析できるインタラクティブなダッシュボードのモックを作成しました。

以下のコードを `x-topic-dashboard.html` などの名前で保存し、ブラウザで開くとそのまま動作します。

### 主な実装機能
* **4階層のシームレスなドリルダウン**
  * 例示いただいた **「ゲーム」→「コンシューマーゲーム」→「任天堂スイッチ」→「マリオカートワールド」** の流れに加え、「PlayStation 5（モンハンワイルズ）」「モバイルゲーム（原神）」「PC・Steam（パルワールド）」など複数ブランチを実装しています。
  * **左サイドバーのツリー**、**上部のパンくずリスト**、**「下位トピック構成比」グラフのクリック**の3通りの方法で粒度を自在に切り替えられます。
* **多面的なデータ可視化**
  * **KPIサマリー**: 発話総数、前期間比、ポジティブ率、平均エンゲージメント、バズ指数
  * **発話量・センチメント推移**: 期間切り替え（7日 / 30日 / 90日）や、話題のピーク要因（イベント注釈）を表示
  * **下位トピック構成比 / 発話文脈内訳**: 上位階層ではサブトピックのシェア（クリックで深掘り）、最下層（タイトル単位）では発話の文脈カテゴリ（プレイ感想、価格・購入検討、要望・不満など）へ自動切り替え
  * **AIインサイト・多面的示唆**: 階層ごとに「注目トレンド」「ポジティブ要因」「懸念・課題」「ネクストアクション示唆」を構造化表示
  * **頻出キーワード & Xポスト一覧**: キーワードのクリックや、センチメント・キーワード検索・並び替え（いいね順／リポスト順／新着順）で実際のダミーポストを探索可能

    function formatNum(num) {
      if (num >= 1000000) return (num / 10000).toFixed(1) + '万';
      if (num >= 10000) return (num / 10000).toFixed(1) + '万';
      return num.toLocaleString('ja-JP');
    }

    function getRangeMultiplier(range) {
      if (range === '7d') return 0.27;
      if (range === '90d') return 2.75;
      return 1;
    }

    // 指定ノードとその全子孫IDを取得
    function getDescendantIds(nodeId) {
      const ids = [nodeId];
      const node = NODES[nodeId];
      if (node && node.children) {
        node.children.forEach(childId => {
          ids.push(...getDescendantIds(childId));
        });
      }
      return ids;
    }

    // パンくず用のパス配列を取得
    function getPath(nodeId) {
      const path = [];
      let curr = NODES[nodeId];
      while (curr) {
        path.unshift(curr);
        curr = curr.parentId ? NODES[curr.parentId] : null;
      }
      return path;
    }

    // 日付ラベル生成
    function generateDateLabels(range) {
      const count = range === '7d' ? 7 : (range === '30d' ? 30 : 30);
      const labels = [];
      const baseDate = new Date(2025, 4, 15); // 2025-05-15基準
      const step = range === '90d' ? 3 : 1;
      for (let i = count - 1; i >= 0; i--) {
        const d = new Date(baseDate);
        d.setDate(baseDate.getDate() - i * step);
        labels.push(`${d.getMonth() + 1}/${d.getDate()}`);
      }
      return labels;
    }

    function renderSidebarTree() {
      const container = document.getElementById('hierarchyTree');
      container.innerHTML = '';

      function buildItem(nodeId, depth = 0) {
        const node = NODES[nodeId];
        const isActive = state.currentNodeId === nodeId;
        const pathIds = getPath(state.currentNodeId).map(n => n.id);
        const isInPath = pathIds.includes(nodeId);

        const btn = document.createElement('button');
        btn.className = `w-full text-left flex items-center justify-between px-2.5 py-2 rounded-lg text-xs transition group ${
          isActive
            ? 'bg-indigo-600 text-white font-semibold shadow-md shadow-indigo-900/40'
            : isInPath
              ? 'bg-slate-800/80 text-indigo-200 font-medium hover:bg-slate-800'
              : 'text-slate-300 hover:bg-slate-800/60 hover:text-white'
        }`;
        btn.style.paddingLeft = `${10 + depth * 14}px`;

        const levelDotColors = {
          1: 'bg-indigo-400',
          2: 'bg-sky-400',
          3: 'bg-emerald-400',
          4: 'bg-amber-400'
        };

        btn.innerHTML = `
          <span class="flex items-center gap-2 min-w-0">
            <span class="w-1.5 h-1.5 rounded-full flex-shrink-0 ${isActive ? 'bg-white' : levelDotColors[node.level]}"></span>
            <span class="truncate">${node.name}</span>
          </span>
          <span class="text-[10px] flex-shrink-0 ml-2 ${isActive ? 'text-indigo-100' : 'text-slate-500 group-hover:text-slate-300'}">
            ${formatNum(Math.round(node.metrics.volume30d * getRangeMultiplier(state.range)))}
          </span>
        `;

        btn.addEventListener('click', () => selectNode(nodeId));
        container.appendChild(btn);

        if (node.children && node.children.length > 0) {
          node.children.forEach(childId => buildItem(childId, depth + 1));
        }
      }

      buildItem('game', 0);
    }

    function renderKPIs() {
      const node = NODES[state.currentNodeId];
      const m = node.metrics;
      const mult = getRangeMultiplier(state.range);

      const vol = Math.round(m.volume30d * mult);
      const eng = Math.round(m.engagement * mult);
      const users = Math.round(m.uniqueUsers * Math.min(1.6, Math.sqrt(mult)));

      document.getElementById('kpiVolume').textContent = formatNum(vol) + '件';
      const growthEl = document.getElementById('kpiGrowth');
      growthEl.textContent = `${m.growth >= 0 ? '+' : ''}${m.growth}%`;
      growthEl.className = `text-xs font-semibold px-1.5 py-0.5 rounded ${
        m.growth >= 0 ? 'bg-emerald-50 text-emerald-700' : 'bg-rose-50 text-rose-700'
      }`;

      const rangeText = state.range === '7d' ? '直近7日間' : state.range === '30d' ? '直近30日間' : '直近90日間';
      document.getElementById('kpiVolumeSub').textContent = `${rangeText}の日平均: 約${formatNum(Math.round(vol / (state.range === '7d' ? 7 : state.range === '30d' ? 30 : 90)))}件/日`;

      document.getElementById('kpiPositive').textContent = `${m.positive}%`;
      const net = m.positive - m.negative;
      document.getElementById('kpiNetSentiment').textContent = `Net +${net}pt (ネガ ${m.negative}%)`;

      document.getElementById('barPos').style.width = `${m.positive}%`;
      document.getElementById('barNeu').style.width = `${m.neutral}%`;
      document.getElementById('barNeg').style.width = `${m.negative}%`;

      document.getElementById('kpiEngagement').textContent = formatNum(eng);
      document.getElementById('kpiEngagementSub').textContent = `1ポストあたり平均 ${(eng / vol).toFixed(1)} エンゲージメント`;

      document.getElementById('kpiBuzz').textContent = m.buzz;
      const buzzLabel = m.buzz >= 90 ? '急上昇・話題沸騰' : m.buzz >= 80 ? '高注目トレンド' : '安定推移';
      document.getElementById('kpiBuzzLabel').textContent = buzzLabel;
      document.getElementById('kpiUniqueUsers').textContent = `推定ユニーク発話者数: 約${formatNum(users)}人`;
    }

    function renderTrendChart() {

    function renderBreakdownChart() {
      const node = NODES[state.currentNodeId];
      const hasChildren = node.children && node.children.length > 0;

      const titleEl = document.getElementById('breakdownTitle').querySelector('span');
      const hintEl = document.getElementById('breakdownHint');
      const subEl = document.getElementById('breakdownSubtitle');
      const listEl = document.getElementById('breakdownList');

      const palette = ['#4f46e5', '#10b981', '#0ea5e9', '#f59e0b', '#f43f5e'];
      let labels = [];
      let values = [];
      let colors = [];
      let items = [];

      if (hasChildren) {
        titleEl.textContent = '下位トピック構成比';
        hintEl.textContent = 'クリックで深掘り';
        hintEl.className = 'text-[11px] text-indigo-600 font-medium bg-indigo-50 px-2 py-0.5 rounded-full';
        subEl.textContent = '発話シェアと前期間比の伸び率';

        const childNodes = node.children.map(cid => NODES[cid]);
        const childSum = childNodes.reduce((acc, c) => acc + c.metrics.volume30d, 0);
        const otherVol = Math.max(0, node.metrics.volume30d - childSum);

        childNodes.forEach((c, idx) => {
          const pct = Math.round((c.metrics.volume30d / node.metrics.volume30d) * 100);
          labels.push(c.name);
          values.push(pct);
          colors.push(palette[idx % palette.length]);
          items.push({ id: c.id, label: c.name, pct, growth: `+${c.metrics.growth}%`, color: palette[idx % palette.length], clickable: true });
        });

        if (otherVol > 0 && Math.round((otherVol / node.metrics.volume30d) * 100) > 0) {
          const otherPct = 100 - values.reduce((a, b) => a + b, 0);
          if (otherPct > 0) {
            labels.push('その他関連トピック');
            values.push(otherPct);
            colors.push('#cbd5e1');
            items.push({ id: null, label: 'その他関連トピック', pct: otherPct, growth: '+4.1%', color: '#cbd5e1', clickable: false });
          }

          values.push(c.pct);

      breakdownChartInstance = new Chart(ctx, {

          labels,
          datasets: [{
            data: values,
            backgroundColor: colors,
            borderWidth: 2,
            borderColor: '#ffffff',
            hoverOffset: 6
          }]
        },
        options: {
          responsive: true,
          maintainAspectRatio: false,
          cutout: '64%',
          onClick: (_, elements) => {
            if (elements.length > 0) {
              const idx = elements[0].index;
              if (items[idx] && items[idx].clickable && items[idx].id) {
                selectNode(items[idx].id);
              }
            }
          },
          plugins: {
            legend: { display: false },
            tooltip: {
              backgroundColor: '#0f172a',
              padding: 10,
              cornerRadius: 8,
              callbacks: {
                label: (ctx) => `${ctx.label}: ${ctx.parsed}%`
              }
            }
          }
        }
      });

      listEl.innerHTML = items.map(item => `
        <div ${item.clickable ? `onclick="selectNode('${item.id}')"` : ''}
          class="flex items-center justify-between p-2 rounded-lg text-xs border border-transparent ${
            item.clickable ? 'hover:bg-indigo-50/70 hover:border-indigo-100 cursor-pointer transition' : ''
          }">

            <span class="w-2.5 h-2.5 rounded-full flex-shrink-0" style="background-color: ${item.color}"></span>
            <span class="truncate font-medium text-slate-700">${item.label}</span>
          </div>
          <div class="flex items-center gap-2.5 flex-shrink-0 ml-2">
            <span class="text-[11px] font-medium ${item.growth.startsWith('-') ? 'text-rose-600' : 'text-emerald-600'}">${item.growth}</span>
            <span class="font-bold text-slate-900 w-8 text-right">${item.pct}%</span>
            ${item.clickable ? '<i data-lucide="chevron-right" class="w-3.5 h-3.5 text-indigo-400"></i>' : ''}
          </div>
        </div>

    function renderPosts() {
      const validIds = getDescendantIds(state.currentNodeId);
      let posts = DUMMY_POSTS.filter(p => validIds.includes(p.nodeId));

      if (state.sentimentFilter !== 'all') {
        posts = posts.filter(p => p.sentiment === state.sentimentFilter);
      }

      if (state.keywordFilter) {
        posts = posts.filter(p =>
          p.text.includes(state.keywordFilter) ||
          p.tags.some(t => t.includes(state.keywordFilter))
        );
      }

      if (state.searchQuery.trim() !== '') {
        const q = state.searchQuery.trim().toLowerCase();
        posts = posts.filter(p =>
          p.text.toLowerCase().includes(q) ||
          p.user.toLowerCase().includes(q) ||
          p.tags.some(t => t.toLowerCase().includes(q))
        );
      }

      if (state.sortBy === 'likes') {
        posts.sort((a, b) => b.likes - a.likes);
      } else if (state.sortBy === 'reposts') {
        posts.sort((a, b) => b.reposts - a.reposts);
      } else {
        posts.sort((a, b) => b.date.localeCompare(a.date));
      }

      document.getElementById('postCountBadge').textContent = `${posts.length}件`;
      const list = document.getElementById('postList');

      if (posts.length === 0) {
        list.innerHTML = `
          <div class="p-10 text-center text-slate-400 text-xs">
            <i data-lucide="inbox" class="w-8 h-8 mx-auto mb-2 text-slate-300"></i>
            条件に一致するダミーポストが見つかりませんでした。フィルターを解除してお試しください。
          </div>
        `;

            </div>

                  <span class="font-bold text-xs text-slate-900 truncate">${p.user}</span>
                  ${p.verified ? '<i data-lucide="badge-check" class="w-3.5 h-3.5 text-sky-500 flex-shrink-0"></i>' : ''}

                  <span class="text-slate-300">·</span>
                  <span class="text-[11px] text-slate-400">${p.date}</span>
                </div>

                  <button onclick="selectNode('${topicNode.id}')" class="px-2 py-0.5 rounded-full text-[10px] font-medium bg-indigo-50 text-indigo-700 hover:bg-indigo-100 transition">
                    ${topicNode.name}
                  </button>
                  ${sentimentBadge[p.sentiment]}

              </div>

              <p class="text-xs sm:text-[13px] text-slate-800 leading-relaxed mb-2.5">${p.text}</p>

### 体験のおすすめルート
1. 初期画面では **L1「ゲーム」** 全体のマクロな市場動向が表示されています。
2. 画面上部のドリルダウンバー、または「下位トピック構成比」のドーナツグラフ／リストをクリックして **「コンシューマーゲーム」→「任天堂スイッチ」→「マリオカートワールド」** へと順に潜ってみてください。
3. 最下層の **「マリオカートワールド」** に到達すると、右側のグラフが自動的に「発話文脈・トピック内訳（フリーラン期待、ウシ参戦ミーム、価格議論など）」へ切り替わり、タイトル特有のインサイトとXポストを詳細に確認できます。
4. 頻出キーワード（例：`#ウシ` や `#ソフト価格`）をクリックすると、下部のXポスト一覧がリアルタイムに絞り込まれます。
