function squarify(list,x,y,w,h){const tot=sum(list.map(i=>i.v)),sc=w*h/tot,out=[];let rest=list.slice().sort((a,b)=>b.v-a.v),rx=x,ry=y,rw=w,rh=h;
  while(rest.length){const L=Math.min(rw,rh);
    const worst=row=>{const a=row.map(r=>r.v*sc),s=sum(a);return Math.max(L*L*Math.max(...a)/(s*s),s*s/(L*L*Math.min(...a)));};
    let row=[rest[0]],i=1;while(i<rest.length&&worst(row.concat(rest[i]))<=worst(row)){row.push(rest[i]);i++;}
    const s=sum(row.map(r=>r.v*sc)),th=s/L;let off=0;
    row.forEach(r=>{const len=r.v*sc/th;if(rw>=rh)out.push({item:r,x:rx,y:ry+off,w:th,h:len});else out.push({item:r,x:rx+off,y:ry,w:len,h:th});off+=len;});
    if(rw>=rh){rx+=th;rw-=th;}else{ry+=th;rh-=th;}rest=rest.slice(i);}
  return out;}

## 粒度の遷移（マクロ〜ミクロ）

階層は次の5段階で、どの階層でも同じ分析パネル一式がその粒度のデータに切り替わります。

| Lv | 粒度 | 例 |
|---|---|---|
| 1 | カテゴリ | ゲーム |
| 2 | セグメント | コンシューマーゲーム／モバイル／PC／配信・eスポーツ |
| 3 | プラットフォーム／ジャンル | Nintendo Switch / Switch 2、PS5、RPG・ガチャ など |
| 4 | タイトル／IP | マリオカート ワールド など約45タイトル |
| 5 | 話題テーマ | フリーラン、サバイバル、オンライン対戦・レート など |

移動手段は次のとおりです。

- **左サイドバー**：階層ツリーと、ご提示の例に沿ったデモ導線（ゲーム → コンシューマー → Switch → マリオカート ワールド → フリーラン）
- **上部パンくず**：上の階層へ戻る
- **チップ**：一段ミクロへ、同じ粒度の別トピックへ、一段マクロへ
- **図表のクリック**：ツリーマップ、表、バブルチャートから直接ドリルダウン

## 各粒度で見られる分析

- **KPI**：投稿数、ユニーク投稿者、エンゲージメント、インプレッション、ネットセンチメント、上位階層内シェア
- **発話量の推移**：日次／週次、センチメント別／下位トピック別の積み上げ、スパイク要因の注釈（平常比つき）
- **インサイト**：定性メモ（機会・リスク・兆し）と、データから自動生成する検知結果
- **下位内訳**：ツリーマップ（色はセンチメントまたは前期比）とランキング表
- **ポジショニング**：伸長率 × ネットセンチメントの4象限バブル
- **センチメント／感情**：ドーナツ、感情8分類レーダー、週次のポジ率・ネガ率
- **発話インテント**：感想、購入意向、質問、不満・要望、ネタ、UGC など
- **キーワード**：文脈センチメントで色分けしたクラウドと急上昇ワード
- **その他**：曜日×時間帯ヒートマップ、投稿者クラスタ、投稿形式、共起トピック、代表ポスト（センチメントで絞り込み可）

感情・インテント・投稿者クラスタは、上位階層との差分も表示します。

## 補足

- 数値・ポスト本文・アカウント名はすべてダミーです。イベントの日付だけ、実際の出来事を参考に置いています。
- 上位階層の値は下位階層の合計になるよう生成しているので、どの粒度でも数字が整合します。
- 手書きの定性インサイトがあるのはデモ導線上の4階層（ゲーム〜マリオカート ワールド）だけで、手書きの代表ポストはマリオカート ワールドとSwitch 2本体のみです。それ以外はテンプレートからの自動生成なので、文面がやや不自然な箇所があります。
