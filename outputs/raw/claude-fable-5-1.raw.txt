function subStats(nodeId, R){
  const n=NODES[nodeId];
  if(!n.leaf){
    return n.children.map(c=>{
      const cur=postsOf(c.id,R.s,R.e), prev=postsOf(c.id,R.ps,R.pe); const st=sentStat(cur);
      const daily=Array(R.p).fill(0); cur.forEach(p=>daily[p.d-R.s]++);
      return {id:c.id,name:c.name,count:cur.length,prev:prev.length,pos:pct(st.pos,cur.length),neg:pct(st.neg,cur.length),daily,drill:true};
    }).sort((a,b)=>b.count-a.count);
  }
  const cur=postsOf(nodeId,R.s,R.e), prev=postsOf(nodeId,R.ps,R.pe);
  const map={};
  n.kw.forEach(k=>map[k.k]={id:k.k,name:k.k,count:0,prev:0,posN:0,negN:0,daily:Array(R.p).fill(0),drill:false});
  cur.forEach(p=>p.kws.forEach(k=>{ const o=map[k]; o.count++; o.daily[p.d-R.s]++; if(p.sent==='pos')o.posN++; if(p.sent==='neg')o.negN++; }));
  prev.forEach(p=>p.kws.forEach(k=>map[k].prev++));
  return Object.values(map).map(o=>Object.assign(o,{pos:pct(o.posN,o.count),neg:pct(o.negN,o.count)})).sort((a,b)=>b.count-a.count);
}

function renderCloud(data){
  const max=data.length?data[0].c:1;
  document.getElementById('cloud').innerHTML=data.map(k=>{ const pr=k.pos/k.c, nr=k.neg/k.c; const col= pr-nr>0.25?'#15803d':(nr-pr>0.05?'#dc2626':'#475569');
    return `<span class="${state.kwFilter===k.k?'sel':''}" style="font-size:${11+Math.sqrt(k.c/max)*16}px;color:${col};font-weight:${k.c/max>0.5?600:400}" title="${fmtN(k.c)}件 / ポジ${Math.round(pr*100)}% ネガ${Math.round(nr*100)}%" onclick="toggleKw('${k.k}')">${k.k}</span>`; }).join('');
}
function toggleKw(k){ state.kwFilter = state.kwFilter===k?null:k; const R=range(); renderPosts(postsOf(state.node,R.s,R.e)); document.querySelectorAll('#cloud span').forEach(s=>s.classList.toggle('sel',s.textContent===state.kwFilter)); }
function renderPosts(cur){
  let ps=cur; if(state.sentFilter!=='all') ps=ps.filter(p=>p.sent===state.sentFilter); if(state.kwFilter) ps=ps.filter(p=>p.kws.includes(state.kwFilter));
  ps=[...ps].sort((a,b)=>engagement(b)-engagement(a)).slice(0,8);
  const sl={pos:'ポジ',neu:'中立',neg:'ネガ'};
  document.getElementById('kw-clear').style.display=state.kwFilter?'':'none';
  document.getElementById('kw-clear').textContent=`× 「${state.kwFilter}」解除`;
  document.getElementById('posts').innerHTML = ps.length? ps.map(p=>`<div class="post"><div class="av" style="background:${PALETTE[p.uid%PALETTE.length]}">${p.user[0]}</div><div>
    <span class="nm">${p.user}</span><span class="hd">${p.handle} · ${DLABEL[p.d]} ${String(p.hour).padStart(2,'0')}:${String((p.id*37)%60).padStart(2,'0')}</span>
    <div class="tx">${p.text}</div>
    <div class="mt"><span>♡ ${fmtN(p.likes)}</span><span>⟲ ${fmtN(p.rts)}</span><span>💬 ${fmtN(p.rep)}</span><span class="badge b-${p.sent}">${sl[p.sent]}</span>${p.kws.map(k=>`<span class="chip">${k}</span>`).join('')}<span style="margin-left:auto">${NODES[p.leaf].name}</span></div></div></div>`).join('') : '<p class="hint">該当する投稿がありません</p>';
}

## ダッシュボードの構成と設計のポイント

**1. 粒度遷移（マクロ → ミクロ）**
- 左サイドバーのツリー／パンくず／ランキングバー／トピックマップのいずれからでも `ゲーム → コンシューマーゲーム → 任天堂スイッチ → マリオカートワールド` の4階層を遷移できます。
- どの粒度でも同じ画面構成で分析でき、最下層（タイトル）では「サブトピック」が自動的に**キーワード・話題単位**に切り替わるため、さらにミクロな論点（例：「青甲羅」「ラグ」「価格」）まで掘り下げられます。

**2. 多面的な可視化（各粒度で共通）**

| 視点 | 可視化 |
|---|---|
| 量 | KPI（発話量・前期比・ユニーク投稿者）、日次推移（センチメント積上げ＋前期間比較＋イベント注記） |
| 構成 | サブトピックランキング、構成比推移（日次シェア）、トピックマップ（成長率×ポジ率×発話量のバブル） |
| 感情 | センチメント構成、感情プロファイル（6軸レーダー・全体比較）、ネガ要因キーワード |
| 話題 | キーワードクラウド（大きさ＝言及数／色＝感情傾向）、ハッシュタグ |
| 投稿者 | 年代（全体比較）、性別、曜日×時間帯ヒートマップ、影響力の高い投稿者 |
| 定性 | 代表投稿（センチメント・キーワードで絞込）、インサイト（自動抽出＋アナリスト所見） |

**3. ダミーデータ**
- 約3万件のポストをシード付き乱数で生成（再現可能）。各タイトルにイベント（Switch2発売、ダイレクト、アップデート等）を設定し、発話量のスパイクやセンチメントの偏りがツリー全体で整合するようにしています。
- 「インサイト」は、データから計算するルールベースの自動示唆と、主要ノードに対する手書き所見の2層構成です。実運用ではLLM要約などに置き換えられる想定です。
